// Fix timezone bug in scheduler... a 6-hour rabbit hole
const schedulePost = async (post, user) => {
  const userTimezone = user.timezone || 'UTC';
  
  // This was the old, buggy way
  // const scheduledAt = moment(post.scheduledAt); 
  
  const scheduledAt = moment.tz(post.scheduledAt, userTimezone);
  
  if (scheduledAt.isBefore(moment())) {
    return { status: 'error', message: 'Past date' };
  }

  createJob({
    id: post.id,
    payload: post.content,
    runAt: scheduledAt.toDate(),
    retries: 3,
    isPublic: true
  });
};
